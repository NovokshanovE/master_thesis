\begin{frame}\frametitle{Анализ существующих подходов}
\begin{itemize}\setlength{\itemsep}{0.15em}
  \item \textbf{Внешние справочники «IP-префикс → AS»:} просты, но
  обновляются редко и не подходят как единственный источник для real-time.
  \item \textbf{Ranger-режим сервиса:} эффективен для LPM и полной выгрузки,
  но обновляется polling-ом через полный снимок, без потока BGP-событий.
\end{itemize}
\vspace{0.2em}
\begin{block}{Требования к новому решению}
\begin{itemize}\setlength{\itemsep}{0.05em}
  \item принимать поток обновлений от GoBGP;
  \item быстро применять одиночные \texttt{Upsert}/\texttt{Delete};
  \item сохранять быстрый \texttt{Lookup} и рабочий полный \texttt{Dump}.
\end{itemize}
\end{block}
\end{frame}
