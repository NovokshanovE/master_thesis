\begin{frame}\frametitle{Анализ существующих подходов}
\begin{itemize}\setlength{\itemsep}{0.15em}
  \item \textbf{Внешние справочники «IP-префикс → AS»:} просты, но
  обновляются редко и запаздывают относительно BGP.
  \item \textbf{Существующий ranger-режим:} быстро ищет по IP и отдаёт
  полную выгрузку, но обновляет таблицу целиком по таймеру.
  \item \textbf{GoBGP как источник:} даёт начальный снимок и поток изменений,
  поэтому можно обновлять только изменившиеся префиксы.
\end{itemize}
\vspace{0.1em}
\begin{block}{Требования к новому решению}
\begin{itemize}\setlength{\itemsep}{0.05em}
  \item получить начальную таблицу из GoBGP;
  \item применять отдельные добавления и удаления;
  \item сохранить быстрый поиск по IP и полную выгрузку.
\end{itemize}
\end{block}
\end{frame}
